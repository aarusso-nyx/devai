const bytes = Buffer.from('{"fixture":"immutable exact bytes","materialized_at":"fixture-time"}\n');
const hash = (value: Uint8Array) => createHash('sha256').update(value).digest('hex');
const payload = () => ({
  kind: 'authority-policy-create-review',
  schema_version: '1.0.0',
  status: 'awaiting-exact-confirmation',

function fixture() {
  let clock = 10;
  const verifyActual = vi.fn();
  const publish = vi.fn((frozen: Uint8Array) => ({ observed_sha256: hash(frozen) }));
  const sourceBytes = Buffer.from(bytes);
  const sourceReady = payload();
  const handle = createAuthorityPolicyReview({
    ready: sourceReady,
    bytes: sourceBytes,
    now: () => clock,
    verifyActual,
    publish,
  });
  return {
    handle,
    sourceBytes,
    sourceReady,
    verifyActual,
    publish,
    setClock: (n: number) => {
      clock = n;
    },
  };
}
interface ReviewReport extends Readonly<Record<string, unknown>> {
  readonly plan_sha256: string;
  readonly artifact: {
    readonly sha256: string;
    readonly bytes_base64: string;
    readonly size_bytes: number;
    readonly encoding: string;
  };
}
function start(handle: ReturnType<typeof createAuthorityPolicyReview>): ReviewReport {
  return handle.start() as ReviewReport;
}
const confirmation = (report: ReviewReport) =>
  `CREATE ${report.plan_sha256} ${report.artifact.sha256}`;

function put(root: string, path: string, content: string | object) {
  const target = join(root, path);
  mkdirSync(dirname(target), { recursive: true });
  writeFileSync(
    target,
    typeof content === 'string' ? content : `${JSON.stringify(content, null, 2)}\n`,
  );
}
function inventory(root: string): Record<string, string> {
  const result: Record<string, string> = {};
  function walk(path: string) {
    for (const entry of readdirSync(join(root, path), { withFileTypes: true })) {
      const name = path ? `${path}/${entry.name}` : entry.name;
      if (entry.isDirectory()) walk(name);
      else if (entry.isFile()) result[name] = hash(readFileSync(join(root, name)));
      else result[name] = `special:${lstatSync(join(root, name)).mode}`;
    }
  }
  walk('');
  return result;
}

      result: { verdict: 'review' },
    });
    const ready = initial[0].result.value as ReviewReport;
    expect(stderr).toContain(`CREATE ${ready.plan_sha256} ${ready.artifact.sha256}`);
    expect(f.rawMutations).not.toHaveBeenCalled();
    expect(inventory(config.root)).toEqual(before);
    // Harness coordination files are outside the immutable repository and package.
    writeFileSync(
      join(config.control, `${config.id}.ready.json`),
      `${JSON.stringify({ label: owner.label, pid: process.pid, id: config.id, ready_envelope: initial[0], prompt: stderr, input_delivered: false, before }, null, 2)}\n`,
    );
    const command = await awaitRaceJson(join(config.control, `${config.id}.confirm.json`));
    expect(command.nonce).toBe(config.nonce);
    expect(typeof command.confirmation).toBe('string');
    terminal.emit('line', command.confirmation);
    await pending;
    const envelopes = [...stdout.split('\n'), ...stderr.split('\n')]
      .filter((line) => line.startsWith('{'))
      .map((line) => JSON.parse(line));
    expect(envelopes).toHaveLength(2);
    const stat = lstatSync(join(config.root, '.devai/config/authority-policy.json'), {
      bigint: true,
    });
    const bytes = readFileSync(join(config.root, '.devai/config/authority-policy.json'));
    const observed = {
      sha256: hash(bytes),
      bytes_base64: bytes.toString('base64'),
      identity: {
        dev: stat.dev.toString(),
        ino: stat.ino.toString(),
        birthtimeNs: stat.birthtimeNs.toString(),
      },
    };
    const result = {
      label: owner.label,
      pid: process.pid,
      id: config.id,
      envelopes,
      stdout,
      stderr,
      exit: process.exitCode ?? 0,
      observed,
      after: inventory(config.root),
    };
    expect(process.stdout.write).toBe(streams.stdout);
    expect(process.stderr.write).toBe(streams.stderr);
    expect(process.exit).toBe(streams.exit);
    disposeCliInvocationAuthority();
    expect(requireAuthorityPolicyReviewOutput).toThrow(
      'AUTHORITY_POLICY_REVIEW_INPUTS_UNSUPPORTED',
    );
    writeFileSync(
      join(config.control, `${config.id}.result.json`),
      `${JSON.stringify(result, null, 2)}\n`,
    );
  } finally {
    terminal.emit('close');
    await pending;
    disposeCliInvocationAuthority();
    f.host.dispose();
    vi.doUnmock('node:readline');
    out.mockRestore();
    err.mockRestore();
    process.argv = saved.argv;
    process.exitCode = saved.exitCode;
  }
}

it(PUBLIC_RACE_TEST, async () => {
  const inherited = process.env['DEVAI_P1_RACE_CHILD_CONFIG'];
  if (inherited !== undefined) {
    await runPublicRaceChild(JSON.parse(inherited) as SharedRaceChild);
    return;
  }
  const f = await productionFixture(true);
  const control = realpathSync(mkdtempSync(join(tmpdir(), 'devai-p1-race-control-')));
  fixtureRoots.push(control);
  const nonce = randomUUID();
  const owner = {
    label: 'SYNTHETIC DISPOSABLE P1-18 RACE',
    root: f.root,
    packageRoot: f.packageRoot,
    nonce,
  };
  writeFileSync(join(control, 'owner.json'), `${JSON.stringify(owner)}\n`);
  const startChild = (id: 'first' | 'second') => {
    const child = spawn(
      process.execPath,
      [
        'scripts/process/validation-runner.mjs',
        '--',
        'node_modules/.bin/vitest',
        'run',
        '--config',
        'tests/config/local.config.ts',
        '--maxWorkers',
        '1',
        'packages/cli/tests/unit/authority-policy-review.test.ts',
        '-t',
        PUBLIC_RACE_TEST,
      ],
      {
        cwd: sourceRoot,
        env: {
          ...process.env,
          GIT_OPTIONAL_LOCKS: '0',
          DEVAI_P1_RACE_CHILD_CONFIG: JSON.stringify({ ...owner, control, id }),
        },
        stdio: ['ignore', 'pipe', 'pipe'],
      },
    );
    let output = '';
    let errors = '';
    let closed = false;
    let code: number | null = null;
    child.stdout.on('data', (chunk: Buffer) => {
      output += chunk.toString();
    });
    child.stderr.on('data', (chunk: Buffer) => {
      errors += chunk.toString();
    });
    const completion = new Promise<void>((done, fail) => {
      child.once('error', fail);
      child.once('close', (exit) => {
        closed = true;
        code = exit;
        done();
      });
    });
    return {
      child,
      completion,
      get closed() {
        return closed;
      },
      get code() {
        return code;
      },
      get log() {
        return output + errors;
      },
      failure: () => (closed ? `owned child ${id} exited ${code}: ${output}${errors}` : undefined),
    };
  };
  const first = startChild('first');
  const second = startChild('second');
  let readyRecords: Record<string, unknown>[] = [];
  let results: Record<string, unknown>[] = [];
  try {
    const before = inventory(f.root);
    readyRecords = await Promise.all([
      awaitRaceJson(join(control, 'first.ready.json'), first.failure),
      awaitRaceJson(join(control, 'second.ready.json'), second.failure),
    ]);
    expect(readyRecords[0]?.pid).not.toBe(readyRecords[1]?.pid);
    expect(readyRecords[0]?.pid).not.toBe(process.pid);
    expect(readyRecords[1]?.pid).not.toBe(process.pid);
    expect(inventory(f.root)).toEqual(before);
    expect(existsSync(join(f.root, '.devai/config/authority-policy.json'))).toBe(false);
    const ready = readyRecords.map((record) => {
      expect(record.input_delivered).toBe(false);
      return (record.ready_envelope as { result: { value: ReviewReport } }).result.value;
    });
    expect(ready[0]?.plan_sha256).not.toBe(ready[1]?.plan_sha256);
    expect(ready[0]?.artifact.sha256).not.toBe(ready[1]?.artifact.sha256);
    const capturedBeforePrecursor = {
      ...owner,
      ready: readyRecords,
      inventory: before,
      precursor_performed: false,
    };
    f.advance();
    const successor = inventory(f.root);
    for (const [index, id] of (['first', 'second'] as const).entries()) {
      const report = ready[index];
      if (!report) throw new Error('missing child ready');
      writeFileSync(
        join(control, `${id}.confirm.json`),
        `${JSON.stringify({ nonce, confirmation: confirmation(report) })}\n`,
      );
    }
    results = await Promise.all([
      awaitRaceJson(join(control, 'first.result.json'), first.failure),
      awaitRaceJson(join(control, 'second.result.json'), second.failure),
    ]);
    await Promise.all([first.completion, second.completion]);
    expect(first.code, first.log).toBe(0);
    expect(second.code, second.log).toBe(0);
    const terminal = (record: Record<string, unknown>) =>
      (record.envelopes as Array<Record<string, unknown>>)[1];
    const winners = results.filter((record) => terminal(record)?.ok === true);
    const losers = results.filter((record) => terminal(record)?.ok === false);
    expect(winners).toHaveLength(1);
    expect(losers).toHaveLength(1);
    const winner = winners[0];
    const loser = losers[0];
    if (!winner || !loser) throw new Error('race outcome missing');
    const winnerReady = readyRecords.find((record) => record.id === winner.id);
    if (!winnerReady) throw new Error('winner ready missing');
    const approved = (winnerReady.ready_envelope as { result: { value: ReviewReport } }).result
      .value;
    const bytes = readFileSync(join(f.root, '.devai/config/authority-policy.json'));
    expect(bytes).toEqual(Buffer.from(approved.artifact.bytes_base64, 'base64'));
    expect(hash(bytes)).toBe(approved.artifact.sha256);
    const stat = lstatSync(join(f.root, '.devai/config/authority-policy.json'), { bigint: true });
    const identity = {
      dev: stat.dev.toString(),
      ino: stat.ino.toString(),
      birthtimeNs: stat.birthtimeNs.toString(),
    };
    expect(winner.observed).toEqual(loser.observed);
    expect(winner.observed).toEqual({
      sha256: hash(bytes),
      bytes_base64: bytes.toString('base64'),
      identity,
    });
    expect((terminal(winner)?.result as { value: Record<string, unknown> }).value).toMatchObject({
      status: 'created',
      binding_performed: true,
      digest_sha256: approved.artifact.sha256,
      publication_identity: identity,
    });
    expect(loser.exit).not.toBe(0);
    const after = inventory(f.root);
    delete after['.devai/config/authority-policy.json'];
    expect(after).toEqual(successor);
    expect(
      readdirSync(join(f.root, '.devai/config')).some((name) => name.endsWith('.publish-staged')),
    ).toBe(false);
    const evidence = process.env['DEVAI_P1_RACE_EVIDENCE_DIR'];
    if (evidence) {
      writeFileSync(
        join(evidence, 'p1-18-both-ready-before-precursor.json'),
        `${JSON.stringify(capturedBeforePrecursor, null, 2)}\n`,
      );
      writeFileSync(
        join(evidence, 'p1-18-race-completion.json'),
        `${JSON.stringify({ ...owner, results, winner: winner.id, loser: loser.id, identity, artifact_sha256: hash(bytes), actual_successor: successor, after: inventory(f.root) }, null, 2)}\n`,
      );
    }
  } finally {
    if (!first.closed) first.child.kill('SIGTERM');
    if (!second.closed) second.child.kill('SIGTERM');
    await Promise.allSettled([first.completion, second.completion]);
    const evidence = process.env['DEVAI_P1_RACE_EVIDENCE_DIR'];
    if (evidence) {
      writeFileSync(join(evidence, 'p1-18-child-first.txt'), first.log);
      writeFileSync(join(evidence, 'p1-18-child-second.txt'), second.log);
    }
    f.host.dispose();
  }
});
