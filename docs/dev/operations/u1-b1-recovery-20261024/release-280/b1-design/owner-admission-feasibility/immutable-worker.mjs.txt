import { DatabaseSync } from 'node:sqlite';
import { readFileSync, writeFileSync } from 'node:fs';
const [path, counter, mode] = process.argv.slice(2);
const db = new DatabaseSync(path, { timeout: mode === 'contender' ? 100 : 2000 });
try {
  db.exec('BEGIN IMMEDIATE');
  if (mode === 'holder') {
    console.log(JSON.stringify({ acquired: true }));
    setInterval(() => {}, 1000);
  } else {
    const before = Number(readFileSync(counter, 'utf8'));
    Atomics.wait(new Int32Array(new SharedArrayBuffer(4)), 0, 0, 75);
    writeFileSync(counter, String(before + 1));
    db.exec('ROLLBACK');
    db.close();
    console.log(JSON.stringify({ acquired: true, before, after: before + 1 }));
  }
} catch (error) {
  console.log(JSON.stringify({ acquired: false, errcode: error.errcode }));
  db.close();
}
