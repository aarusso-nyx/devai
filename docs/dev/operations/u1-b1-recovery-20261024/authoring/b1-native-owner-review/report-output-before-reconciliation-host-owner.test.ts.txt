function fault(callback: () => void) {
  const before = readHooks.observe;

    callback();
  } finally {
    readHooks.observe = before;
  }

const hash = (bytes: string | Uint8Array) => createHash('sha256').update(bytes).digest('hex');
function jcs(value: unknown): string {
  if (value === null || typeof value !== 'object') return JSON.stringify(value);
  if (Array.isArray(value)) return '[' + value.map(jcs).join(',') + ']';
  const object = value as Data;
  return (
    '{' +
    Object.keys(object)
      .sort()
      .map((key) => JSON.stringify(key) + ':' + jcs(object[key]))
      .join(',') +
    '}'
  );
}

function inventory(path: string): unknown {
  if (!existsSync(path)) return null;
  const s = lstatSync(path, { bigint: true });
  if (s.isSymbolicLink()) return { kind: 'symlink', ino: String(s.ino), mode: String(s.mode) };
  return s.isDirectory()
    ? {
        kind: 'directory',
        ino: String(s.ino),
        mode: String(s.mode),
        entries: readdirSync(path)
          .sort()
          .map((name) => [name, inventory(resolve(path, name))]),
      }
    : {
        kind: 'file',
        ino: String(s.ino),
        mode: String(s.mode),
        size: String(s.size),
        sha256: hash(readFileSync(path)),
      };
}
function issue(api: RuntimeApi, issuer: AuthorityDecisionIssuer, target: Data) {
  const action = 'init bind',
    role = 'architect';
  const rule = {
    rule_id: 'native-provision-fixture',
    origin: 'additive-extension',
    precedence: 500,
    action_ids: [action],
    selector: {
      kind: target.kind,
      repository_id: target.repository_id,
      operations: [target.operation],
    },
    effect: 'allow',
    subjects: [{ kind: 'human', roles: [role] }],
    required_consent: CONSENT,
    constitutional_anchors: [6],
    rationale: 'Owned actual native fixture; deterministic B5 is not installed origin.',
  };
  const plant = makePolicyPlant({ additiveRules: [rule] });
  expect(getValidator('authority-policy.schema.json')(plant.document)).toBe(true);
  const context = expectSuccess<{ context_receipt: unknown }>(
    api.resolveAuthorityDeclaration(
      {
        action_id: action,
        invocation_id: 'invocation-1',
        dry_run: false,
        declaration: { as_role: role },
        consent: CONSENT,
      },
      declarationDependencies(
        issuer,
        actionDocumentWithId(action, 'local-write', { kind: 'human', allowed_roles: [role] }),
        undefined,
        policyBindingFromPlant(plant),
      ),
    ),
  );
  const policy = expectSuccess(api.loadAuthorityPolicy({ document: plant.document }, plant.deps));
  const resolution = api.resolveAuthorityPolicy(
    policy,
    {
      action_id: action,
      context_receipt: context.context_receipt,
      consent: CONSENT,
      resource: target,
      operation: target.operation,
    },
    { receiptStore: issuer, canonicalSha256 },
  );
  expect(resolution).toMatchObject({ outcome: 'allow' });
  const base = exactSubject([target], plant) as { plan: { envelope: MutationEnvelope } };
  const envelope = {
    ...base.plan.envelope,
    request: { ...base.plan.envelope.request, action_id: action },
  } as MutationEnvelope;
  const subject = {
    plan: {
      ...base.plan,
      envelope: {
        ...envelope,
        issued_by: {
          ...envelope.issued_by,
          envelope_digest_sha256: computeMutationEnvelopeDigest(envelope),
        },
      },
    },
  };
  const decision = issuer.issueAllow({
    resolutions: [resolution],
    subject,
    context_receipt: context.context_receipt,
    invocation_id: 'invocation-1',
    boundary_adapter_id: 'reconciliation-host-authority-boundary',
  }) as { issued: boolean; receipt: unknown };
  expect(decision.issued).toBe(true);
  return { subject, receipt: decision.receipt };
}

  options: { linked?: boolean; write?: boolean; repo?: ReturnType<typeof repository> } = {},
) {
  const api = await runtimeApi();
  const module = await provisionApi();
  const repo = options.repo ?? repository(options.linked);
  const issuer = createIssuer(api);
  const write = options.write ?? true;
  process.argv = [
    process.execPath,
    'devai',
    'init',
    'bind',
    '--reconciliation-host',
    '--target',
    repo.root,
    '--as-role',
    'architect',
    ...(write ? ['--write'] : []),
    '--format',
    'json',
  ];
  for (const stream of [process.stdin, process.stdout, process.stderr])
    Object.defineProperty(stream, 'isTTY', { value: true, configurable: true });
  const bindings = {
    package_sha256: 'a'.repeat(64),
    module_sha256: 'b'.repeat(64),
    registry_sha256: 'c'.repeat(64),
    constitution_sha256: 'd'.repeat(64),
    policy_sha256: 'e'.repeat(64),
  };
  let currentBindings = bindings;
  const argvSha = canonicalSha256(process.argv);
  const invocation = createReconciliationHostInvocation({
    invocation_id: 'invocation-1',
    action_id: 'init bind',
    receipt_store: issuer,
    repository_root: repo.root,
    repository_id: 'example-repository',
    original_argv_sha256: argvSha,
    initiator: 'architect',
    operation: 'provision',
    phases: [],
    write,
    require_tty: write,
    readCurrentBindings: () => currentBindings,
    verifyActual: () => {
      if (canonicalSha256(process.argv) !== argvSha) throw new Error('argv drift');
    },
  });
  const targets: Data[] = [];
  const scope = {
    action_id: 'init bind',
    invocation_id: 'invocation-1',
    effect: 'local-write' as const,
    receipt_store: issuer,
    reconciliation_invocation: invocation,
    apply_effect: (request: AuthorityHostEffectRequest, apply: () => unknown) => {
      const view = reconciliationHostOperation(request, invocation);
      expect(view).toBeDefined();
      if (!view) throw new Error('missing actual native operation');
      const target = view.target as unknown as Data;
      targets.push(target);
      const grant = issue(api, issuer, target);
      let result: unknown;
      const adapter = {
        admit: (candidate: Data) =>
          reconciliationHostOperation(request, invocation) === view &&
          canonicalSha256(candidate) === canonicalSha256(target),
        snapshot: () => reconciliationHostOperationSnapshot(view, invocation, view.target),
        apply: () => {
          result = apply();
          return result;
        },
      };
      const runtime = createAuthorityBoundaryRuntime(
        boundaryDependencies(issuer, [], { reconciliation_host: adapter }),
      );
      const handle = expectSuccess(
        runtime.plannerRegistry.registerPlan({
          subject: grant.subject,
          invocation_id: 'invocation-1',
        }),
      );
      const prepared = expectSuccess(
        runtime.prepare({
          subject: grant.subject,
          plan_handle: handle,
          target,
          decision_receipt: grant.receipt,
          adapter_id: 'reconciliation-host-authority-boundary',
        }),
      );
      expectSuccess(runtime.apply({ prepared }));
      return result;
    },
  };
  return {
    module,
    repo,
    scope,
    targets,
    bindings,
    currentBindings: () => currentBindings,
    run: <T>(callback: () => T) => runWithAuthorityHostEffects(scope, callback),
    setBindings: (value: typeof bindings) => {
      currentBindings = value;
    },
    dispose: () => {
      disposeReconciliationHostInvocation(invocation);
      issuer.dispose();
    },
  };
}

      ['grant-000002.json', renewal],
      ['grant-000003.json', revocation],
      ['intent-01.json', intent],
      ['phase-01.json', completion],
      ['lease.json', lease],
    ]);
  };
  const install = (id = OWNER_ID, graphInvalid = false) => {
    const parent = resolve(f.repo.namespace, 'approvals');
    if (!existsSync(parent)) mkdirSync(parent, { mode: 0o700 });
    const path = resolve(parent, id);
    mkdirSync(path, { mode: 0o700 });
    const map = records(id, graphInvalid);
    for (const [name, record] of map) {
      expect(getValidator('reconciliation-owner.schema.json')(record)).toBe(true);
      writeFileSync(resolve(path, name), encoded(record), { mode: 0o600 });
    }
    return { path, map };
  };
  const selected = install();
  return {
    ...f,
    issuer,
    issuerBytes,
    key,
    keyBytes,
    initial,
    selected,
    records,
    install,
    encoded,
    signed,
    dispose: () => {
      keyBytes.fill(0);
      issuerBytes.fill(0);
      f.dispose();
    },
  };
}
function readScope(
  f: Awaited<ReturnType<typeof fixture>>,
  options: {
    action?: string;
    selector?: string;
    write?: boolean;
    noInvocation?: boolean;
    forged?: boolean;
  } = {},
) {
  const action = options.action ?? 'evidence reconcile approve',
    selector = options.selector ?? OWNER_ID,
    write = options.write ?? false;
  const phaseAction = action !== 'evidence reconcile approve' && action !== 'init bind';
  const initiator =
    action === 'sense bootstrap-inventory' || action === 'init bind'
      ? 'architect'
      : action === 'evidence reconcile approve'
        ? 'owner'
        : 'inspector';
  process.argv = [
    process.execPath,
    'devai',
    ...action.split(' '),
    '--approval',
    selector,
    '--plan-sha256',
    PLAN,
    '--as-role',
    initiator,
    ...(write ? ['--write'] : []),
  ];
  const argv = canonicalSha256(process.argv);
  const invocation = createReconciliationHostInvocation({
    action_id: action,
    invocation_id: 'invocation-1',
    receipt_store: f.scope.receipt_store,
    repository_root: f.repo.root,
    repository_id: 'example-repository',
    original_argv_sha256: argv,
    initiator,
    operation: action === 'init bind' ? 'provision' : phaseAction ? 'consume-phase' : 'renew',
    phases:
      action === 'evidence reconcile materialize'
        ? ['01', '02']
        : action === 'evidence reconcile stage-proof'
          ? ['03']
          : action === 'sense bootstrap-inventory'
            ? ['04']
            : [],
    write,
    require_tty: write && !phaseAction,
    ...(action === 'init bind' ? {} : { approval_id: selector, plan_sha256: PLAN }),
    verifyActual: () => {
      if (canonicalSha256(process.argv) !== argv) throw new Error('fixture argv drift');
    },
    readCurrentBindings: f.currentBindings,
  });
  let effects = 0;
  const scope = {
    action_id: action,
    invocation_id: 'invocation-1',
    effect: 'local-write' as const,
    receipt_store: f.scope.receipt_store,
    ...(options.noInvocation
      ? {}
      : { reconciliation_invocation: options.forged ? {} : invocation }),
    apply_effect: () => {
      effects++;
      throw new Error('NO_FIXTURE_READ_EFFECT_AUTHORITY');
    },
  };
  return {
    scope,
    invocation,
    effects: () => effects,
    run: <T>(callback: () => T) => runWithAuthorityHostEffects(scope, callback),
    dispose: () => disposeReconciliationHostInvocation(invocation),
  };
}
function expectSnapshot(
  snapshot: Data,
  f: Awaited<ReturnType<typeof fixture>>,
  expectedNamespace: Data,
  expectedApproval: Data,
  matches = true,
) {
  safe(snapshot, f);
  expect(Object.keys(snapshot).sort()).toEqual(
    [
      'kind',
      'identity',
      'current_binding_sha256',
      'namespace_sha256',
      'approval_namespace_sha256',
      'binding_matches_approval',
      'clock',
      'state',
      'grants_authority',
    ].sort(),
  );
  expect(snapshot.kind).toBe('reconciliation-host-approval-snapshot');
  expect(snapshot.identity).toEqual(f.initial.identity);
  expect(snapshot.current_binding_sha256).toBe(hash(jcs(f.currentBindings())));
  expect(snapshot.namespace_sha256).toBe(hash(jcs(expectedNamespace)));
  expect(snapshot.approval_namespace_sha256).toBe(hash(jcs(expectedApproval)));
  expect(snapshot.binding_matches_approval).toBe(matches);
  expect(snapshot.grants_authority).toBe(false);
  expect(freezeData(snapshot)).toBe(true);
  const at = snapshot.clock as Data;
  expect(at.clock_source).toBe(f.initial.clock.clock_source);
  expect(at.boot_id).toBe(f.initial.clock.boot_id);
  expect(Number(at.wall_ms) >= f.initial.clock.wall_ms).toBe(true);
  expect(BigInt(String(at.monotonic_ns)) >= BigInt(f.initial.clock.monotonic_ns)).toBe(true);
}
