function canonical(value: unknown): string {
  if (value === null || typeof value !== 'object') return JSON.stringify(value);
  if (Array.isArray(value)) return '[' + value.map(canonical).join(',') + ']';
  return (
    '{' +
    Object.keys(value)
      .sort()
      .map((k) => JSON.stringify(k) + ':' + canonical((value as Record<string, unknown>)[k]))
      .join(',') +
    '}'
  );
}

function inventory(dir: string): unknown[] {
  return readdirSync(dir, { withFileTypes: true })
    .sort((a, b) => a.name.localeCompare(b.name))
    .flatMap((entry) => {
      const p = resolve(dir, entry.name),
        stat = lstatSync(p, { bigint: true });
      const row = {
        path: p.slice(dir.length + 1),
        dev: String(stat.dev),
        ino: String(stat.ino),
        mode: String(stat.mode),
        size: String(stat.size),
        mtime: String(stat.mtimeNs),
        ctime: String(stat.ctimeNs),
        bytes: entry.isFile()
          ? digest(readFileSync(p))
          : entry.isSymbolicLink()
            ? readlinkSync(p)
            : null,
      };
      return entry.isDirectory()
        ? [row, ...inventory(p).map((item) => ({ directory: entry.name, item }))]
        : [row];
    });
}
function directoryIdentity(path: string): Record<string, string> {
  const s = lstatSync(path, { bigint: true });
  return {
    dev: String(s.dev),
    ino: String(s.ino),
    birthtime_ns: String(s.birthtimeNs),
    uid: String(s.uid),
    mode: String(s.mode & 0o7777n),
  };
}
function fileIdentity(path: string): Record<string, string> {
  const stat = lstatSync(path, { bigint: true });
  return {
    ...directoryIdentity(path),
    size_bytes: String(stat.size),
    mtime_ns: String(stat.mtimeNs),
    ctime_ns: String(stat.ctimeNs),
  };
}
function identityProjection(root: string): Record<string, unknown> {
  const git_dir = realpathSync(git(root, ['rev-parse', '--absolute-git-dir'])),
    common_dir = realpathSync(
      git(root, ['rev-parse', '--path-format=absolute', '--git-common-dir']),
    );
  const entry = resolve(root, '.git'),
    s = lstatSync(entry, { bigint: true });
  return {
    repository_id: 'fixture-repository',
    repository_root: root,
    git_dir,
    common_dir,
    root_identity: directoryIdentity(root),
    git_dir_identity: directoryIdentity(git_dir),
    common_dir_identity: directoryIdentity(common_dir),
    git_entry_identity: s.isDirectory()
      ? directoryIdentity(entry)
      : {
          ...directoryIdentity(entry),
          size_bytes: String(s.size),
          mtime_ns: String(s.mtimeNs),
          ctime_ns: String(s.ctimeNs),
        },
    git_entry_sha256: s.isDirectory() ? null : digest(readFileSync(entry)),
    git_backlink_identity: s.isDirectory() ? null : fileIdentity(resolve(git_dir, 'gitdir')),
    git_backlink_sha256: s.isDirectory() ? null : digest(readFileSync(resolve(git_dir, 'gitdir'))),
    commondir_identity: s.isDirectory() ? null : fileIdentity(resolve(git_dir, 'commondir')),
    commondir_sha256: s.isDirectory() ? null : digest(readFileSync(resolve(git_dir, 'commondir'))),
  };
}

function run(
  mode: string,
  root: string,
  variant = '',
  executable = process.execPath,
  extraEnv: Record<string, string> = {},
): Reply {
  writeFileSync(worker, workerSource);
  const before = process.hrtime.bigint();
  const result = spawnSync(executable, [worker, mode, root, variant], {
    env: { ...cleanEnv, ...extraEnv },
    encoding: 'utf8',
    timeout: 4500,
    maxBuffer: 1024 * 1024,
  });
  const after = process.hrtime.bigint();
  childReceipts.push({
    mode,
    root,
    variant,
    executable,
    pid: result.pid,
    exit: result.status,
    signal: result.signal,
    stdout: result.stdout,
    stderr: result.stderr,
    parent_before_ns: String(before),
    parent_after_ns: String(after),
  });
  expect(result.signal, result.stderr).toBeNull();
  expect(result.status, result.stderr).toBe(0);
  return JSON.parse(result.stdout.trim()) as Reply;
}

    });
  for (const variant of [
    'head',
    'duplicate',
    'no-LF',
    'uppercase',
    'wrong-format',
    'too-many',
    'bad-hex',
    'oversize',
  ])
    it(`refuses ${variant} MERGE_HEAD bytes`, () => {

      if (variant === 'oversize') text = 'b'.repeat(4097);
      writeFileSync(resolve(admin, 'MERGE_HEAD'), text);
      expect(run('capture', f.root).error.code).toBe('RECONCILIATION_HOST_INVALID');
    });
  it('observes valid merge heads in file order without modifying the merge', () => {
    const f = fixture();

    const before = inventory(f.base);
    expect(run('basic', f.root).identity.merge_heads).toEqual(parents);
    expect(inventory(f.base)).toEqual(before);
  });
  it('never opens private issuer/seed contents while observing an existing secure namespace', () => {

describe('isolated negative producer and sampling failures without changing OS clocks', () => {

    'boot-stale',
    'runtime-stale',
    'sample-width',

  ])

          : ['boot-stale', 'runtime-stale'].includes(variant)

            : 'RECONCILIATION_HOST_CLOCK_UNAVAILABLE',
      );

      expect(JSON.stringify(value.error)).not.toContain('SECRET_');
    });
  for (const variant of ['root', 'repository'])
    it(`does not coerce Proxy ${variant} input`, () => {

      expect(value.error.code).toBe('RECONCILIATION_HOST_INVALID');
      expect(value.touches).toBe(0);
    });
  it('membership precedes environment refusal and environment-only failure does not silently dispose the real member', () => {
    const value = run('fault', fixture().root, 'member-env');
    expect(value.results[0]?.error?.code).toBe('RECONCILIATION_HOST_CONTEXT_UNKNOWN');
    expect(value.results[1]?.error?.code).toBe('RECONCILIATION_HOST_ENVIRONMENT_INVALID');
    expect(value.results[2]?.error).toBeUndefined();
  });
  it('actual pass-through probes have fixed tools, flags, clean environment and native bounds', () => {
    const value = run('basic', fixture().root);
    const gitProbes = value.probes.filter((p) => p.file === '/usr/bin/git');
    expect(gitProbes.length).toBeGreaterThanOrEqual(7);
    for (const probe of gitProbes) {
      for (const flag of [
        '--no-optional-locks',
        '--no-replace-objects',
        '--no-lazy-fetch',
        'core.fsmonitor=false',
      ])

        uv: string;

      };
      expect(observed).toMatchObject({
        node: process.versions.node,
